    """Map sync URL onto async driver"""
    if url.startswith("postgresql://"):
        return url.replace("postgresql://", "postgresql+asyncpg://", 1)
    if url.startswith("sqlite://"):
        return url.replace("sqlite://", "sqlite+aiosqlite://", 1)
    return url

def _strip_sslmode(url: str) -> tuple[str, Optional[str]]:
    """Extract sslmode (libpq keyword) before passing to asyncpg
    (asyncpg.connect() rejects 'sslmode' param)"""
    parsed = make_url(url)
    sslmode = parsed.query.get("sslmode")
    if sslmode:
        query = {k: v for k, v in parsed.query.items() if k != "sslmode"}
        return parsed.set(query=query).render_as_string(...), sslmode
    return url, None
