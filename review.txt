def test_build_fit_cost_lookup_ai_identifier_shape():
    """Test build_fit_cost_lookup with AI Identifier shaped output."""
    saved_output = {
        "Cost Saving Potential": {
            "l1_l2_heatmap": [
                {
                    "L1_name": "Finance & Accounting",
                    "L1_percent": 85,
                    "L1_amount": 250000,
                    "L2s": [
                        {
                            "L2_name": "General Ledger",
                            "L2_percent": 90,
                            "L2_amount": 120000,
                        },
                        {
                            "L2_name": "Tax Processing",
                            "L2_percent": 80,
                            "L2_amount": 130000,
                        },
                    ],
                },
            ],
            "impact_analyzer": [
                {
                    "L1 Capability": "Finance & Accounting",
                    "L2 Capability": "General Ledger",
                    "L3 Capability": "Journal Entry Processing",
                    "High Fit %": 95,
                    "Cost Saving Opportunity": 60000,
                },
                {
                    "L1 Capability": "Finance & Accounting",
                    "L2 Capability": "Tax Processing",
                    "L3 Capability": "Tax Return Compilation",
                    "High Fit %": 75,
                    "Cost Saving Opportunity": 45000,
                },
            ],
        },
    }

    lookup, ambiguous = build_fit_cost_lookup(saved_output)

    assert ("finance & accounting",) in lookup
    assert lookup[("finance & accounting",)]["high_fit_percent"] == 85
    assert lookup[("finance & accounting",)]["cost_saving_opportunity"] == 250000

    assert ("finance & accounting", "general ledger") in lookup
    assert lookup[("finance & accounting", "general ledger")]["high_fit_percent"] == 90
    assert lookup[("finance & accounting", "general ledger")]["cost_saving_opportunity"] == 120000

    assert ("finance & accounting", "general ledger", "journal entry processing") in lookup
    assert lookup[("finance & accounting", "general ledger", "journal entry processing")]["high_fit_percent"] == 95
    assert lookup[("finance & accounting", "general ledger", "journal entry processing")]["cost_saving_opportunity"] == 60000

    assert ambiguous == []
