def _extract_fit_cost_block(prompt: str) -> str:
    """Extract the CAPABILITY FIT AND COST-SAVING CONTEXT block from the prompt."""
    start_marker = "CAPABILITY FIT AND COST-SAVING CONTEXT"
    end_marker = "RUBRIC"
    
    start_idx = prompt.find(start_marker)
    if start_idx == -1:
        return ""
    
    end_idx = prompt.find(end_marker, start_idx)
    if end_idx == -1:
        return ""
    
    return prompt[start_idx:end_idx].strip()
