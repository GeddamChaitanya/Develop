`_extract_impact_analyzer_lookup(blob)` takes an optional output dictionary and returns a dictionary keyed by normalized L3 capability name. It looks first under `Cost Saving Potential.impact_analyzer`, then falls back to root-level `impact_analyzer`.

```python
def _extract_impact_analyzer_lookup(blob: Optional[Dict[str, Any]]) -> Dict[str, Dict[str, Any]]:
    """Extract impact_analyzer rows keyed by normalized L3 capability name."""
    if not isinstance(blob, dict):
        return {}

    # Both WFA and OPEX pipelines store: {"Cost Saving Potential": {"impact_analyzer": [...]}}
    csp = blob.get("Cost Saving Potential", {})
    ia_rows: list = csp.get("impact_analyzer", []) if isinstance(csp, dict) else []
    if not ia_rows:
        # Fallback: impact_analyzer at the root level (OPEX GENERATE_ALL)
        ia_rows = blob.get("impact_analyzer", [])
    if not isinstance(ia_rows, list) or not ia_rows:
        return {}

    lookup: Dict[str, Dict[str, Any]] = {}
    for ia in ia_rows:
        if not isinstance(ia, dict):
            continue
        # WFA uses "L3 Capability", OPEX uses "L3 Capability" or "L3 Capability Name (Lvl 3)"
        name = (
            ia.get("L3 Capability")
            or ia.get("L3 Capability Name (Lvl 3)")
            or ia.get("l3CapabilityName")
            or ""
        )
        key = _normalise_capability_name(name)
        if key:
            lookup[key] = ia
    return lookup
```

`_load_impact_analyzer_for_process(process_key)` queries recent cost-potential outputs for that process, tries inline JSON and then blob storage, and returns the first nonempty lookup. It returns `{}` if it finds no usable output.

```python
async def _load_impact_analyzer_for_process(process_key: int) -> Dict[str, Dict[str, Any]]:
    """Load impact_analyzer rows from the most recent Cost Saving Potential
    output that belongs to the same blueprinting process.

    Returns a dict keyed by **normalised L3 capability name** (lower-cased,
    stripped) whose values are the full impact-analyzer row dicts.  Returns
    an empty dict if no matching output is found.
    """
    from db.database import AsyncSessionLocal
    from sqlalchemy import text as sa_text

    db = None
    try:
        db = AsyncSessionLocal()
        # Match BOTH naming conventions:
        #   WFA:  "Workforce Cost Saving Potential Output for action ..."
        #   OPEX: "OPEX Generation Output for action COST_SAVING_POTENTIAL"
        rows = (await db.execute(sa_text(
            "SELECT o.output_json, o.output_file_storage_name, o.output_file_path, o.output_key, o.output_name "
            "FROM blueprinting_process_sub_step_output o "
            "JOIN blueprinting_process_sub_step ss ON o.sub_step_key = ss.sub_step_key "
            "JOIN blueprinting_process_step s  ON ss.step_key  = s.step_key "
            "WHERE s.process_key = :pk "
            "  AND (o.output_name LIKE :wfa_pattern "
            "       OR o.output_name LIKE :opex_pattern) "
            "ORDER BY o.created_date DESC LIMIT 10"
        ), {
            "pk": process_key,
            "wfa_pattern": "%Cost Saving Potential%",
            "opex_pattern": "%COST_SAVING_POTENTIAL%",
        })).fetchall()

        if not rows:
            return {}

        for row in rows:
            try:
                blob = _parse_json_payload(row[0])
                if blob is None:
                    blob = _load_json_payload_from_storage(row[1] or row[2])

                lookup = _extract_impact_analyzer_lookup(blob)
                if lookup:
                    return lookup
            except Exception as row_exc:
                track_trace(
                    f"Warning: could not parse Cost Saving Potential output "
                    f"for process_key={process_key}, output_key={row[3] if len(row) > 3 else 'unknown'}: {row_exc}",
                    severity="Warning",
                )

        return {}
    except Exception as exc:
        track_trace(
            f"Warning: could not load impact_analyzer for process_key={process_key}: {exc}",
            severity="Warning",
        )
        return {}
    finally:
        if db is not None:
            await db.close()
```

`_enrich_capabilities_from_impact_analyzer(l3_caps, impact_lookup)` mutates the input L3 dictionaries in place and returns the count enriched. A capability with no normalized-name match is skipped unchanged.

```python
def _enrich_capabilities_from_impact_analyzer(
    l3_caps: List[dict],
    impact_lookup: Dict[str, Dict[str, Any]],
) -> int:
    """Merge authoritative financial fields from the impact_analyzer onto each
    L3 capability dict (in-place).  Returns the number of capabilities that
    were enriched.

    The impact_analyzer is THE source of truth for the Heatmap, so using it
    here ensures the KPI Impact tab shows identical dollar figures.
    """
    # Map: impact-analyzer field name  →  L3 capability target field name.
    # Both WFA and OPEX field variants are listed; whichever exists in the
    # impact-analyzer row will be written.
    FIELD_MAP: List[tuple] = [
        # (ia_field_name, target_field_name)
        ("Labor %",                     "Labor_%"),
        ("Labor Amount (USD)",          "Labor_Opex"),         # WFA
        ("Labor Opex",                  "Labor_Opex"),         # OPEX
        ("High Fit Labor (USD)",        "High_Fit_Labor"),     # WFA
        ("High Fit Labor",              "High_Fit_Labor"),     # OPEX
        ("Low Fit Labor (USD)",         "Low_Fit_Labor"),      # WFA
        ("Low Fit Labor",               "Low_Fit_Labor"),      # OPEX
        ("Discount %",                  "Discount_%"),
        ("Cost Saving Opportunity (USD)", "Cost_Saving_Opportunity"),  # WFA
        ("Cost Saving Opportunity",     "Cost_Saving_Opportunity"),    # OPEX
        ("Total Salary (USD)",          "total_opex"),         # WFA
        ("Total Opex",                  "total_opex"),         # OPEX
        ("High Fit %",                  "High_Fit_%"),
        ("Low Fit %",                   "Low_Fit_%"),
    ]

    enriched = 0
    for cap in l3_caps:
        name = _normalise_capability_name(cap.get("l3CapabilityName"))
        if not name or name not in impact_lookup:
            continue

        ia_row = impact_lookup[name]
        any_set = False
        for ia_field, target_field in FIELD_MAP:
            val = ia_row.get(ia_field)
            if val is not None:
                cap[target_field] = val
                any_set = True
        if any_set:
            enriched += 1

    return enriched
```
